# QtTest-based test suite. One executable per suite; every executable
# registers itself with CTest via add_test().

find_package(Qt6 6.4 REQUIRED COMPONENTS Test)

# The in-memory paired transport is shared by several test executables, so
# build it once into a small support library.
add_library(mcp_test_support STATIC in_memory_transport.cpp)
target_link_libraries(mcp_test_support PUBLIC mcp::mcp Qt6::Test)
target_include_directories(mcp_test_support PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_BINARY_DIR}")

# The plugin_test suite loads a plugin at runtime. Build it here as its own
# MODULE target (independent of any example) so the test is self-contained.
add_library(mcp_test_plugin MODULE test_plugin.cpp)
set_target_properties(mcp_test_plugin PROPERTIES
    PREFIX ""
    OUTPUT_NAME "mcp_test_plugin"
)
target_link_libraries(mcp_test_plugin PRIVATE mcp::mcp)
target_compile_options(mcp_test_plugin PRIVATE
    $<$<OR:$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>,$<CXX_COMPILER_ID:GNU>>:
        -Wall -Wextra -Wpedantic>
)

# A second plugin exercising cross-plugin calls (TestPluginPair::forward →
# TestPlugin::echo through the host registry).
add_library(mcp_test_plugin_pair MODULE test_plugin_pair.cpp)
set_target_properties(mcp_test_plugin_pair PROPERTIES
    PREFIX ""
    OUTPUT_NAME "mcp_test_plugin_pair"
)
target_link_libraries(mcp_test_plugin_pair PRIVATE mcp::mcp)
target_compile_options(mcp_test_plugin_pair PRIVATE
    $<$<OR:$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>,$<CXX_COMPILER_ID:GNU>>:

# Subprocess tests need the absolute path of the example binary they spawn.
# Bake the path into a generated header rather than relying on CTest
# ENVIRONMENT plumbing, which different CMake versions split inconsistently.
# The header is conditional on TARGET availability; tests #ifdef around a
# missing binary.
set(_test_paths_h "${CMAKE_CURRENT_BINARY_DIR}/test_binary_paths.hpp")
file(GENERATE OUTPUT "${_test_paths_h}" CONTENT
"// Generated by tests/CMakeLists.txt — do not edit.
#pragma once

$<$<TARGET_EXISTS:calculator_server>:#define MCP_TEST_CALCULATOR_SERVER \"$<TARGET_FILE:calculator_server>\">
$<$<TARGET_EXISTS:mcp_notes_server>:#define MCP_TEST_NOTES_SERVER \"$<TARGET_FILE:mcp_notes_server>\">
$<$<TARGET_EXISTS:mcp_test_plugin>:#define MCP_TEST_PLUGIN \"$<TARGET_FILE:mcp_test_plugin>\">
$<$<TARGET_EXISTS:mcp_test_plugin_pair>:#define MCP_TEST_PLUGIN_PAIR \"$<TARGET_FILE:mcp_test_plugin_pair>\">
$<$<TARGET_EXISTS:mcp_test_log_plugin>:#define MCP_TEST_LOG_PLUGIN \"$<TARGET_FILE:mcp_test_log_plugin>\">
")

# mcp_add_test(<name> <sources...>): build <name> as a QtTest executable and
# register it with CTest.
function(mcp_add_test name)
    add_executable(${name} ${ARGN})
    target_link_libraries(${name} PRIVATE mcp_test_support Qt6::Test)
    if(MCP_ENABLE_HTTP)
        # mcp::http brings in mcp + Qt Network/HttpServer transitively.
        target_link_libraries(${name} PRIVATE mcp::http)
    endif()
    target_include_directories(${name} PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_BINARY_DIR}")
    target_compile_options(${name} PRIVATE
        $<$<OR:$<CXX_COMPILER_ID:Clang>,$<CXX_COMPILER_ID:AppleClang>,$<CXX_COMPILER_ID:GNU>>:
            -Wall -Wextra -Wpedantic>
    )
    add_test(NAME ${name} COMMAND ${name})
endfunction()
